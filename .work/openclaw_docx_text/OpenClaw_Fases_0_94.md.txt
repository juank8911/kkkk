import os, asyncio, logging
from firecrawl import FirecrawlApp
import redis.asyncio as redis
logger = logging.getLogger("OpenClaw.Scraper")
firecrawl_app = FirecrawlApp(api_key=os.getenv("FIRECRAWL_API_KEY", "demo"))
redis_client = redis.Redis(host='redis-cache', port=6379, db=0)
LIMITES_WAF = {"zillow.com": 15, "realtor.com": 20, "redfin.com": 25, "default": 30}
async def adquirir_permiso_extraccion(dominio: str) -> bool:
limite_rpm = LIMITES_WAF.get(dominio, LIMITES_WAF["default"])
clave_redis = f"ratelimit:openclaw:{dominio}"
async with redis_client.pipeline(transaction=True) as pipe:
try:
pipe.incr(clave_redis)
pipe.expire(clave_redis, 60, nx=True)
resultados = await pipe.execute()
if resultados[0] > limite_rpm:
return False
return True
except:
return True
async def extraer_con_reintentos(url: str, query: str, max_intentos: int = 3):
dominio_base = url.replace("https://", "").replace("www.", "").split("/")[0]
if not await adquirir_permiso_extraccion(dominio_base):
return {"dominio": url, "status": "rate_limited"}
intento = 0
while intento < max_intentos:
try:
scrape_options = {"formats": ["markdown"], "waitFor": 3000, "proxy": True}
res = await asyncio.to_thread(
firecrawl_app.search,
query=f"site:{url} {query}",
scrape_options=scrape_options
)
return {"dominio": url, "status": "exito", "data": res}
except Exception as e:
intento += 1
await asyncio.sleep(2 ** intento)
return {"dominio": url, "status": "fallo_critico"}

import json
async def consumir_cola_scraping():
try:
await redis_client.xgroup_create("openclaw_scraping_queue", "scrapers", mkstream=True)
except:
pass
while True:
try:
mensajes = await redis_client.xreadgroup(
groupname="scrapers", consumername="worker-1",
streams={"openclaw_scraping_queue": ">"}, count=1, block=5000
)
for stream, alertas in mensajes:
for id_mensaje, datos in alertas:
payload = json.loads(datos[b'payload'].decode('utf-8'))
tareas = [extraer_con_reintentos(url, payload["query"]) for url in payload["plataformas"]]
await asyncio.gather(*tareas)
await redis_client.xack("openclaw_scraping_queue", "scrapers", id_mensaje)
except Exception as e:
await asyncio.sleep(2)

def nodo_analisis_financiero(state: dict) -> dict:
precio = state.get("precio_usd", 0)
ingreso = state.get("ingreso_anual_estimado", 0)
if precio <= 0 or ingreso <= 0: return {"estado": "RECHAZADO"}
gastos = (ingreso * 0.08) + (ingreso * 0.10) + (precio * 0.02) + 1200
noi = ingreso - gastos
cap_rate = (noi / precio) * 100
if cap_rate < 8.0: return {"estado": "RECHAZADO"}
return {"estado": "ANALISIS_FINANCIERO_OK", "cap_rate": cap_rate, "noi": noi}
