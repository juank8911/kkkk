import asyncio, logging, json
from app.core.redis_client import redis_client
from app.services.scraper import firecrawl_app
async def monitorear_propiedades_en_oferta():
while True:
claves_activas = await redis_client.keys("openclaw:tracker:activa:*")
for clave in claves_activas:
datos = json.loads(await redis_client.get(clave))
res = await asyncio.to_thread(firecrawl_app.scrape_url, datos["url"], params={"formats": ["markdown"]})
texto = res.get("markdown", "").lower()
if "pending" in texto or "sold" in texto:
# Enviar alerta telegram de cancelación
await redis_client.delete(clave)
await asyncio.sleep(43200)

# FASE 98: WEBSOCKET TELEMETRÍA (app/api/routes/websockets.py)
# ---------------------------------------------------------
from fastapi import APIRouter, WebSocket
router = APIRouter(prefix="/v1/stream")
@router.websocket("/live-feed")
async def endpoint_telemetria_ws(websocket: WebSocket):
await websocket.accept()
pubsub = redis_client.pubsub()
await pubsub.subscribe("openclaw:eventos_asimov")
while True:
mensaje = await pubsub.get_message(ignore_subscribe_messages=True, timeout=1.0)
if mensaje: await websocket.send_text(mensaje["data"].decode("utf-8"))
await asyncio.sleep(0.1)
