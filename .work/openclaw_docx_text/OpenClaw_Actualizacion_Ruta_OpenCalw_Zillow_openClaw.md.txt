import json
import logging
from fastapi import APIRouter, Depends, HTTPException
from app.core.redis_client import redis_client
from app.api.middleware.security import validar_acceso_zero_trust
logger = logging.getLogger("OpenClaw.Ingestion")
router = APIRouter(prefix="/v1/openclaw")
@router.post("/iniciar-barrido", dependencies=[Depends(validar_acceso_zero_trust)])
async def encolar_peticion_scraping(payload: dict):
"""
Recibe la orden de búsqueda (query y plataformas) vía API y la inyecta
en Redis Streams para su consumo asíncrono en el worker de Firecrawl.
"""
query = payload.get("query")
plataformas = payload.get("plataformas", [])
if not query or not plataformas:
logger.warning("Petición rechazada: payload incompleto.")
raise HTTPException(status_code=400, detail="Se requiere 'query' y 'plataformas'.")
mensaje = {"query": query, "plataformas": plataformas}
try:
await redis_client.xadd(
name="openclaw_scraping_queue",
fields={"payload": json.dumps(mensaje)}
)
logger.info(f"Vector de extracción encolado para {len(plataformas)} plataformas.")
return {
"status": "Autorizado y Encolado",
"mensaje": "Coordenadas enviadas al enjambre Firecrawl."
}
except Exception as e:
logger.error(f"Fallo crítico al conectar con el broker Redis: {e}")
raise HTTPException(status_code=500, detail="Error interno de orquestación.")

2. Fase 117 — Circuit Breaker Adaptativo y Evasión WAF (app/core/waf_circuit_breaker.py)
import json
import logging
from app.core.redis_client import redis_client
logger = logging.getLogger("OpenClaw.WAFResilience")
MAX_FAILURES_BEFORE_STEALTH = 2
MAX_FAILURES_BEFORE_TRIP = 4
COOLDOWN_SECONDS = 900
async def obtener_perfil_evasion(plataforma: str) -> dict | None:
clave_abierta = f"openclaw:waf:circuit_open:{plataforma}"
if await redis_client.exists(clave_abierta):
ttl = await redis_client.ttl(clave_abierta)
logger.warning(f"[{plataforma}] Circuito ABIERTO. Enfriamiento restante: {ttl}s.")
return None
fallos = int(await redis_client.get(f"openclaw:waf:failures:{plataforma}") or 0)
if fallos >= MAX_FAILURES_BEFORE_STEALTH:
return {
"formats": ["markdown"],
"onlyMainContent": True,
"waitFor": 3500,
"headers": {"Accept-Language": "en-US,en;q=0.9"}
}
return {"formats": ["markdown"], "onlyMainContent": True}
async def registrar_resultado_waf(plataforma: str, exito: bool):
clave_fallos = f"openclaw:waf:failures:{plataforma}"
if exito:
await redis_client.delete(clave_fallos)
return
fallos_actuales = await redis_client.incr(clave_fallos)
await redis_client.expire(clave_fallos, 1800)
if fallos_actuales >= MAX_FAILURES_BEFORE_TRIP:
await redis_client.set(f"openclaw:waf:circuit_open:{plataforma}", "1", ex=COOLDOWN_SECONDS)
await redis_client.publish(
"openclaw_live_telemetry",
json.dumps({
"plataforma": plataforma,
"estado": "CIRCUITO_ABIERTO",
"cooldown_seg": COOLDOWN_SECONDS
})
)
logger.error(f"[{plataforma}] WAF agresivo detectado. Circuito abierto por {COOLDOWN_SECONDS}s.")

3. Worker Asíncrono de Ingestión en 9 Plataformas (app/workers/ingestion_worker.py)
import asyncio
import json
import logging
import os
from firecrawl import FirecrawlApp
from app.core.redis_client import redis_client
from app.core.waf_circuit_breaker import obtener_perfil_evasion, registrar_resultado_waf
logger = logging.getLogger("OpenClaw.Worker")
firecrawl = FirecrawlApp(api_key=os.getenv("FIRECRAWL_API_KEY"))
PLATAFORMAS_MAP = {
"zillow": "https://www.zillow.com/homes/{query}_rb/",
"realtor": "https://www.realtor.com/realestateandhomes-search/{query}",
"redfin": "https://www.redfin.com/city/{query}",
"trulia": "https://www.trulia.com/{query}",
"loopnet": "https://www.loopnet.com/search/commercial-real-estate/{query}/for-sale/",
"craigslist": "https://{query}.craigslist.org/search/rea",
"century21": "https://www.century21.com/real-estate/{query}/",
"coldwell": "https://www.coldwellbankerhomes.com/{query}/",
"remax": "https://www.remax.com/homes-for-sale/{query}"
}
async def procesar_plataforma(plataforma: str, query: str):
url_base = PLATAFORMAS_MAP.get(plataforma.lower())
if not url_base:
return {"error": "Plataforma no soportada"}
params_evasion = await obtener_perfil_evasion(plataforma)
if params_evasion is None:
return {plataforma: None, "estado": "OMITIDA_POR_CIRCUIT_BREAKER"}
url_target = url_base.format(query=query.replace(" ", "-"))
await redis_client.publish(
"openclaw_live_telemetry",
json.dumps({"plataforma": plataforma, "estado": "ATACANDO_WAF", "query": query})
)
try:
resultado = await asyncio.to_thread(
firecrawl.scrape_url,
url_target,
params=params_evasion
)
await registrar_resultado_waf(plataforma, exito=True)
await redis_client.publish(
"openclaw_live_telemetry",
json.dumps({"plataforma": plataforma, "estado": "BRECHA_CONFIRMADA", "bytes": len(str(resultado))})
)
return {plataforma: resultado}
except Exception as e:
await registrar_resultado_waf(plataforma, exito=False)
await redis_client.publish(
"openclaw_live_telemetry",
json.dumps({"plataforma": plataforma, "estado": "BLOQUEO_WAF", "error": str(e)})
)
return {plataforma: None, "error": str(e)}
async def consumir_cola():
logger.info("Worker OpenClaw iniciado. Escuchando openclaw_scraping_queue...")
while True:
try:
mensajes = await redis_client.xread({"openclaw_scraping_queue": "0-0"}, count=1, block=5000)
if not mensajes:
continue
for stream, msgs in mensajes:
for msg_id, msg_data in msgs:
payload = json.loads(msg_data[b"payload"].decode("utf-8"))
query = payload["query"]
plataformas = payload["plataformas"]
tareas = [procesar_plataforma(p, query) for p in plataformas]
await asyncio.gather(*tareas)
await redis_client.xdel("openclaw_scraping_queue", msg_id)
except Exception as e:
logger.error(f"Error crítico en Worker: {e}")
await asyncio.sleep(5)
if __name__ == "__main__":
asyncio.run(consumir_cola())

4. Fase 116 — Feed WebSocket de Telemetría (app/api/routes/telemetry.py)
import asyncio
import logging
from fastapi import APIRouter, WebSocket, WebSocketDisconnect
from app.core.redis_client import redis_client
logger = logging.getLogger("OpenClaw.Telemetry")
router = APIRouter(prefix="/v1/openclaw/ws", tags=["Telemetry"])
@router.websocket("/scraping-feed")
async def scraping_telemetry_feed(websocket: WebSocket):
await websocket.accept()
pubsub = redis_client.pubsub()
await pubsub.subscribe("openclaw_live_telemetry")
try:
while True:
mensaje = await pubsub.get_message(ignore_subscribe_messages=True, timeout=1.0)
if mensaje:
await websocket.send_text(mensaje["data"].decode("utf-8"))
else:
await asyncio.sleep(0.1)
except WebSocketDisconnect:
logger.info("JeiKeiStation desconectado del feed.")
finally:
await pubsub.unsubscribe("openclaw_live_telemetry")
